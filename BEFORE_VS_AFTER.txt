    FILE: public/app.js
    ═══════════════════

    ❌ BEFORE:
    ────────
    function saveSegment() {
      const blob = new Blob(recordedChunks, { type: chooseMimeType() });
      downloadRecording(blob, filename);  // ❌ Downloads locally!
      recordedChunks = [];
    }

    ✅ AFTER:
    ────────
    async function saveSegment() {
      const blob = new Blob(recordedChunks, { type: chooseMimeType() });
      await uploadToServer(blob, filename);  // ✅ Uploads to server!
      recordedChunks = [];
    }

    async function uploadToServer(blob, filename) {
      const formData = new FormData();
      formData.append('recording', blob, filename);
      
      const response = await fetch('/api/upload', {
        method: 'POST',
        headers: { 'X-Upload-Password': UPLOAD_PASSWORD },
        body: formData
      });
      
      return response.json();
    }
